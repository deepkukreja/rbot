cmake_minimum_required(VERSION 3.16)
project(coupled_mppi_probe)
find_package(ament_cmake REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(xtensor REQUIRED)
find_package(xsimd REQUIRED)
add_executable(mppi_probe mppi_probe.cpp)
target_compile_features(mppi_probe PRIVATE cxx_std_17)
ament_target_dependencies(mppi_probe nav2_mppi_controller)
target_compile_definitions(mppi_probe PRIVATE XTENSOR_ENABLE_XSIMD XTENSOR_USE_XSIMD)
target_compile_options(mppi_probe PRIVATE -O3 -ffast-math -msse4.2 -mavx2 -mfma -mno-avx512f)
target_link_libraries(mppi_probe xtensor xtensor::use_xsimd)
