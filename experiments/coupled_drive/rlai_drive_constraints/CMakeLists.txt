cmake_minimum_required(VERSION 3.16)
project(rlai_drive_constraints)
find_package(ament_cmake REQUIRED)
add_library(rlai_drive_constraints INTERFACE)
target_compile_features(rlai_drive_constraints INTERFACE cxx_std_17)
target_include_directories(rlai_drive_constraints INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
install(TARGETS rlai_drive_constraints EXPORT export_rlai_drive_constraints)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_rlai_drive_constraints)
ament_export_include_directories(include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(wheel_envelope_test test/wheel_envelope_test.cpp)
  target_link_libraries(wheel_envelope_test rlai_drive_constraints)
  target_compile_options(wheel_envelope_test PRIVATE -O2 -ffast-math)
  add_test(NAME wheel_envelope_test COMMAND wheel_envelope_test)
endif()
ament_package()
